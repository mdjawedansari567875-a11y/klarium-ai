// --- Send push notification to every student that has a saved push token --- // Uses Expo's free push notification service directly from the browser — // no backend/server needed. Tokens come from users/{uid}.expoPushToken, // saved by the app itself (pushTokenService.js) after the student grants // notification permission. document.getElementById('notifySendBtn').onclick = async () => { const title = document.getElementById('notifyTitle').value.trim(); const body = document.getElementById('notifyDesc').value.trim(); const statusEl = document.getElementById('notifyStatus'); const btn = document.getElementById('notifySendBtn'); if (!title || !body) { statusEl.textContent = 'Please fill both Title and Description.'; return; } const tokens = allUsers .map((u) => u.expoPushToken) .filter((t) => typeof t === 'string' && t.startsWith('ExponentPushToken')); if (tokens.length === 0) { statusEl.textContent = 'No students have notifications enabled yet.'; return; } btn.disabled = true; statusEl.textContent = `Sending to ${tokens.length} student(s)...`; const chunks = []; for (let i = 0; i < tokens.length; i += 100) { chunks.push(tokens.slice(i, i + 100)); } let sentCount = 0; let failedChunks = 0; for (const chunk of chunks) { const messages = chunk.map((token) => ({ to: token, title, body, sound: 'default', })); try { const res = await fetch('https://exp.host/--/api/v2/push/send', { method: 'POST', headers: { 'Content-Type': 'application/json', }, body: JSON.stringify(messages), }); if (res.ok) { sentCount += chunk.length; } else { failedChunks += 1; } } catch (e) { failedChunks += 1; } } btn.disabled = false; if (sentCount > 0 && failedChunks === 0) { statusEl.textContent = `Sent to ${sentCount} student(s) ✓`; document.getElementById('notifyTitle').value = ''; document.getElementById('notifyDesc').value = ''; } else if (sentCount > 0) { statusEl.textContent = `Sent to ${sentCount} student(s), but some failed — check your network and try again if needed.`; } else { statusEl.textContent = 'Failed to send — please check your internet connection (try switching to Wi-Fi or a stronger signal) and try again.'; } };
